```python
#!/usr/bin/env python3
"""Split one narration take into per-beat WAVs at its silences.

Read the whole script in one go, pause between beats, then cut at the middle of each
pause. One TTS call per video instead of one per beat.

    python3 vo_split.py takes/one-take.wav --out takes/_vo/gameshelf/gameshelf --expect 3
"""
import argparse
import os
import re
import subprocess
import sys


def duration(path):
    out = subprocess.run(
        ["ffprobe", "-v", "error", "-show_entries", "format=duration",
         "-of", "default=nw=1:nk=1", path],
        capture_output=True, text=True, check=True).stdout.strip()
    return float(out)


def silences(path, noise="-35dB", min_s=0.55):
    """Return [(start, end), ...] for every detected silence."""
    p = subprocess.run(
        ["ffmpeg", "-v", "info", "-i", path, "-af",
         f"silencedetect=noise={noise}:d={min_s}", "-f", "null", "-"],
        capture_output=True, text=True)
    starts = [float(m) for m in re.findall(r"silence_start: ([0-9.]+)", p.stderr)]
    ends = [float(m) for m in re.findall(r"silence_end: ([0-9.]+)", p.stderr)]
    return list(zip(starts, ends))


def split(path, out_prefix, expect=None, pad=0.04):
    total = duration(path)
    gaps = silences(path)
    cuts = [0.0] + [round((s + e) / 2, 3) for s, e in gaps] + [total]
    n = len(cuts) - 1
    if expect and n != expect:
        raise ValueError(
            f"expected {expect} parts but found {n} "
            f"({len(gaps)} silence(s) of >=0.55 s detected in {path})")
    os.makedirs(os.path.dirname(os.path.abspath(out_prefix)), exist_ok=True)
    outs = []
    for i in range(n):
        a = max(0.0, cuts[i] - pad)
        b = min(total, cuts[i + 1] + pad)
        out = f"{out_prefix}-{i + 1:02d}.wav"
        subprocess.run(
            ["ffmpeg", "-y", "-v", "error", "-i", path, "-ss", f"{a:.3f}",
             "-to", f"{b:.3f}", "-ar", "48000", "-ac", "1", out], check=True)
        outs.append(out)
    return outs


if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("wav")
    ap.add_argument("--out", required=True)
    ap.add_argument("--expect", type=int, default=None)
    a = ap.parse_args()
    try:
        for p in split(a.wav, a.out, a.expect):
            print(f"{p}  {duration(p):.2f}s")
    except ValueError as exc:
        print(f"error: {exc}", file=sys.stderr)
        sys.exit(1)
```

## Risks, tradeoffs, open questions

1. **Length.** Auto-drafts are 20–30 s, not 90. Length must come from real beats; padding
   a 30 s cut to 90 s produces the exact slack the gate exists to prevent. Target 30–60 s
   and treat longer as a bonus, not a requirement — a tight 40 s beats a padded 90 s.
2. **TTS sameness across seven videos.** One provider voice for seven entries reads as
   one product, which is fine for a brand but weak for distinct entries. The user's own
   voice for the three strongest entries is the highest-quality upgrade available and
   costs only re-recording.
3. **Wayland capture.** Only browser footage is reliable. Any entry whose proof lives in
   a native window (Aura, the desktop itself) gets terminal footage or a still — never a
   black frame passed off as a screen recording.
4. **Deadline arithmetic.** Seven entries × ~15 min ≈ 1 h 45 m of production, plus ~30 min
   of Task 1 writing, plus review. Do Task 1 and Task 2 first: both are prerequisites for
   every entry.
5. **MoA cost.** Enabling it on the openrouter aggregator bills every tool call in the
   loop at Opus rates. It buys a second opinion on narration, which Task 7a gets for free
   from the local lanes.
6. **Open question — entry #8.** `hewn-video-studio` itself is a legitimate entry. Decide
   before Task 8; if yes it needs its own spec and web footage at `http://127.0.0.1:8756/`.
7. **Open question — the seventh entry.** Only six install dirs were scanned; the design
   studio lives in `~/Development/hermes-design-studio` and ships through a different
   path. Scan it explicitly rather than assuming it behaves like the plugins.
